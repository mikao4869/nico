# 조건

# 코드 



js 


    const choseInput = document.querySelector("#chose-form input");
    const guessInput = document.querySelector("#guess-form input");
    const guessButton = document.querySelector("#guess-form button");
    const resultDisplay = document.createElement("div"); 
    document.body.appendChild(resultDisplay);  //result  생각하지 못함 

    choseInput.addEventListener("input", () => {
      const choseNumber = parseInt(choseInput.value, 10);
      if (!isNaN(choseNumber) && choseNumber > 0) {
        guessInput.max = choseNumber;
      }
    });    //조건 3을 참고하여 추가한 코드 

    function handleGuessButton() {
      const choseNumber = parseInt(choseInput.value, 10);  //value는 string이기 때문에 그거를 number로 바꾸기 위해서 parseInt를 사용
      const guessNumber = parseInt(guessInput.value, 10);

      const machineNumber = Math.floor(Math.random() * (choseNumber + 1));  //내가 지정한 숫자 내에서 랜덤숫자 고르는거 

      if (choseNumber > 0) {  // 조건 2 번을 생각해서 한거 
        if (isNaN(guessNumber) || guessNumber < 0 || guessNumber > choseNumber) {
          resultDisplay.innerText = "Error";
          return;   // guessNumber에 아무것도 없거나 guessNumber가 0보다 작거나guessNumber가 choseNumber보다 크거나 그러면 에러 
        }

        resultDisplay.innerText = "You chose: " + guessNumber + " , the machine chose: " + machineNumber;

        if (guessNumber === machineNumber) {
          resultDisplay.innerText += "\nYou Won!";  //guessNumber와 machineNumber가 같으면 이김 
        } else {
          resultDisplay.innerText += "\nYou Lost!";  //지면 짐
        }
      } 
    }

    guessButton.addEventListener("click", handleGuessButton);  //이벤트

# 2. 실시간으로 범위 값 업데이트 

+ choseInput.addEventListener("input", () => {...})

choseInput 요소에서 input 이벤트(입력값이 바뀔 때마다)가 발생하면 콜백 함수 실행.
const choseNumber = parseInt(choseInput.value, 10);

choseInput의 현재 입력값을 정수(parseInt)로 변환하여 choseNumber에 저장.
if (!isNaN(choseNumber) && choseNumber > 0) { ... }

isNaN(choseNumber): 숫자가 아닌 값(NaN)인지 확인.
choseNumber > 0: 0보다 큰 값인지 확인.
즉, 입력된 값이 숫자이고, 0보다 클 때만 아래 코드 실행.
guessInput.max = choseNumber;

guessInput(추측값 입력창)의 max 속성을 choseNumber 값으로 설정.
즉, 사용자가 choseInput에 10을 입력하면 guessInput의 최대 입력 가능 값(max)도 10으로 변경됨.



        choseInput.addEventListener("input", () => {

            if (!isNaN(choseNumber) && choseNumber > 0) {
              guessInput.max = choseNumber;
            }
          });
