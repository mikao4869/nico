# 목표

버튼누르면 랜덤으로 배경색이 바뀌는거





# 코드 

      const colors = [
        '#ef5777',
        '#575fcf',
        '#4bcffa',
        '#34e7e4',
        '#0be881',
        '#f53b57',
        '#3c40c6',
        '#0fbcf9',
        '#00d8d6',
        '#05c46b',
        '#ffc048',
        '#ffdd59',
        '#ff5e57',
        '#d2dae2',
        '#485460',
        '#ffa801',
        '#ffd32a',
        '#ff3f34',
      ];  //색깔 정리 코드 

      const body=document.querySelector("body");
      const button=document.querySelector("button");

      function click(){
        const color1=colors[Math.floor(Math.random()*colors.length)];
        const color2=colors[Math.floor(Math.random()*colors.length)];
        body.style.background= `linear-gradient(to left,${color1},${color2})`;

      }


      button.addEventListener("click",click);


# 설명

단순한 코드였지만 내가 하지 못한거:


+ event: button 누르면 색깔 변해야한다는것을 인식하였지만 하지는 못함
- css에서 `linear-gradient를 사용함` 그러지 말고 js 에서 body.style.background해서 지정했어야했음


+ linear-gradient(색깔1,색깔2) : 이게 그라데이션 코드인데, 색깔 1과 색깔2  가 그라데이션하면서 나옴
난 `to left`를 사용하였는데 이거는 그라데이션이 왼쪽에서부터 시작됨
